! =====================================================================
! R1 - SEDE CENTRAL (192.168.2.0/28 usuarios + .64/28 invitados)
! Laboratorio 13 - Sesion B (independiente): VLANs + DHCP + OSPF
! Pegar en la CLI del router: modo exec -> enable -> configure terminal
! (o pegar directo desde el prompt R1>)
! =====================================================================

enable
configure terminal
!
! --- Identificacion del equipo ---
hostname R1-SEDE-CENTRAL
no ip domain-lookup
enable secret cisco
service password-encryption
!
banner motd @ R1-SEDE-CENTRAL - Lab 13 Sesion B OSPF @

! --- Lineas de acceso ---
line console 0
 password cisco
 logging synchronous
line vty 0 4
 password cisco
 login
 transport input ssh
!
! --- LAN usuarios (VLAN 10) - gateway de la sede central ---
interface GigabitEthernet0/0
 description SEDE CENTRAL - Usuarios VLAN10 GW 192.168.2.1/28
 ip address 192.168.2.1 255.255.255.240
 no shutdown
!
! --- LAN invitados (VLAN 20) - router-on-a-stick ---
interface GigabitEthernet0/0.20
 description SEDE CENTRAL - Invitados VLAN20 GW 192.168.2.65/28
 encapsulation dot1Q 20
 ip address 192.168.2.65 255.255.255.240
 no shutdown
!
! --- UPLINK a SUCURSAL-1 (R2) - camino primario de SUC-1 ---
interface GigabitEthernet0/1
 description Uplink SUC-1 - Red 192.168.2.48/30 (R1=.49, R2=.50)
 ip address 192.168.2.49 255.255.255.252
 no shutdown
!
! --- UPLINK a SUCURSAL-2 (R3) - camino primario de SUC-2 ---
interface GigabitEthernet0/2
 description Uplink SUC-2 - Red 192.168.2.56/30 (R3=.57, R1=.58)
 ip address 192.168.2.58 255.255.255.252
 no shutdown
!
! =========================================================
! DHCP SERVER por VLAN
! VLAN 10: excluidas .1 (gateway) y .2-.5 (fijas admin)
!          pool asignable: .6 - .14
! VLAN 20: excluidas .65 (gateway) y .66-.69 (fijas admin)
!          pool asignable: .70 - .78
! (lease no existe en el 2911 de Packet Tracer 9.0.1: el alquiler
!  por defecto de 1 dia se aplica solo.)
! =========================================================
ip dhcp excluded-address 192.168.2.1 192.168.2.5
ip dhcp pool LAN-CENTRAL
 network 192.168.2.0 255.255.255.240
 default-router 192.168.2.1
 dns-server 8.8.8.8
 domain-name lab-utp.pa
exit
!
ip dhcp excluded-address 192.168.2.65 192.168.2.69
ip dhcp pool VLAN20-CENTRAL
 network 192.168.2.64 255.255.255.240
 default-router 192.168.2.65
 dns-server 8.8.8.8
 domain-name lab-utp.pa
exit
!
! =========================================================
! OSPF AREA 0 - unico protocolo (no hay rutas estaticas)
! =========================================================
router ospf 10
 router-id 192.168.2.1
 network 192.168.2.0 0.0.0.255 area 0
 passive-interface GigabitEthernet0/0
 passive-interface GigabitEthernet0/0.20
!
! =========================================================
! Alternativa RIPv2 (para comparar en la Parte 9)
! =========================================================
! no router ospf 10
! router rip
!  version 2
!  no auto-summary
!  network 192.168.2.0
!
end
copy running-config startup-config
