! =====================================================================
! R2 - SEDE B (LAN-B 192.168.2.16/28)
! Laboratorio 13 - VLSM + DHCP + Enrutamiento
! Pegar en la CLI del router: modo exec -> enable -> configure terminal
! (o pegar directo desde el prompt R2>)
! =====================================================================

enable
configure terminal
!
! --- Identificacion del equipo ---
hostname R2-SEDE-B
no ip domain-lookup
enable secret cisco
service password-encryption
!
banner motd @ R2-SEDE-B - Laboratorio 13 VLSM/DHCP/OSPF @

! --- Lineas de acceso ---
line console 0
 password cisco
 logging synchronous
line vty 0 4
 password cisco
 login
 transport input ssh
!
! --- Interfaz LAN-B (gateway de la sede) ---
interface GigabitEthernet0/0
 description LAN-B - Gateway 192.168.2.17/28
 ip address 192.168.2.17 255.255.255.240
 no shutdown
!
! --- Enlace punto a punto con R1 ---
interface GigabitEthernet0/1
 description P2P R1-R2 - Red 192.168.2.48/30
 ip address 192.168.2.50 255.255.255.252
 no shutdown
!
! --- Enlace punto a punto con R3 ---
interface GigabitEthernet0/2
 description P2P R2-R3 - Red 192.168.2.52/30
 ip address 192.168.2.53 255.255.255.252
 no shutdown
!
! =========================================================
! DHCP SERVER para LAN-B
! Excluidas: .17 (gateway) y .18-.21 (IP fija administrativa)
! Pool asignable: 192.168.2.22 - 192.168.2.30
! =========================================================
ip dhcp excluded-address 192.168.2.17 192.168.2.21
!
ip dhcp pool LAN-B
 network 192.168.2.16 255.255.255.240
 default-router 192.168.2.17
 dns-server 8.8.8.8 1.1.1.1
 domain-name lab-utp.pa
 lease 1
exit
!
! =========================================================
! FASE A - ENRUTAMIENTO ESTATICO
! LAN-A (.0/28) via R1 (.49) | LAN-C (.32/28) via R3 (.54)
! =========================================================
ip route 192.168.2.0 255.255.255.240 192.168.2.49
ip route 192.168.2.32 255.255.255.240 192.168.2.54
!
! =========================================================
! FASE B - OSPF AREA 0 (desactivar la Fase A antes)
! =========================================================
! router ospf 10
!  network 192.168.2.0 0.0.0.255 area 0
!  passive-interface GigabitEthernet0/0
!  no ip route 192.168.2.0 255.255.255.240 192.168.2.49
!  no ip route 192.168.2.32 255.255.255.240 192.168.2.54
!
! =========================================================
! FASE B (alternativa) - RIPv2
! =========================================================
! no router ospf 10
! router rip
!  version 2
!  no auto-summary
!  network 192.168.2.0
!
end
copy running-config startup-config
