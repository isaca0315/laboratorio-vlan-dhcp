! =====================================================================
! R2 - SUCURSAL 1 (192.168.2.16/28 usuarios + .80/28 invitados)
! Laboratorio 13 - Sesion B (independiente): VLANs + DHCP + OSPF
! Pegar en la CLI del router: modo exec -> enable -> configure terminal
! (o pegar directo desde el prompt R2>)
! =====================================================================

enable
configure terminal
!
! --- Identificacion del equipo ---
hostname R2-SUCURSAL-1
no ip domain-lookup
enable secret cisco
service password-encryption
!
banner motd @ R2-SUCURSAL-1 - Lab 13 Sesion B OSPF @

! --- Lineas de acceso ---
line console 0
 password cisco
 logging synchronous
line vty 0 4
 password cisco
 login
 transport input ssh
!
! --- LAN usuarios (VLAN 10) - gateway de la sucursal 1 ---
interface GigabitEthernet0/0
 description SUCURSAL-1 - Usuarios VLAN10 GW 192.168.2.17/28
 ip address 192.168.2.17 255.255.255.240
 no shutdown
!
! --- LAN invitados (VLAN 20) - router-on-a-stick ---
interface GigabitEthernet0/0.20
 description SUCURSAL-1 - Invitados VLAN20 GW 192.168.2.81/28
 encapsulation dot1Q 20
 ip address 192.168.2.81 255.255.255.240
 no shutdown
!
! --- UPLINK a R1 (sede central) - camino primario ---
interface GigabitEthernet0/1
 description Uplink a CENTRAL - Red 192.168.2.48/30 (R1=.49, R2=.50)
 ip address 192.168.2.50 255.255.255.252
 no shutdown
!
! --- RESPALDO a R3 (sucursal 2) - costo alto, solo emergencia ---
interface GigabitEthernet0/2
 description Respaldo a SUC-2 - Red 192.168.2.52/30 (R2=.53, R3=.54)
 ip address 192.168.2.53 255.255.255.252
 ip ospf cost 100
 no shutdown
!
! =========================================================
! DHCP SERVER por VLAN
! VLAN 10: excluidas .17 (gateway) y .18-.21 (fijas admin)
!          pool asignable: .22 - .30
! VLAN 20: excluidas .81 (gateway) y .82-.85 (fijas admin)
!          pool asignable: .86 - .94
! =========================================================
ip dhcp excluded-address 192.168.2.17 192.168.2.21
ip dhcp pool LAN-SUC1
 network 192.168.2.16 255.255.255.240
 default-router 192.168.2.17
 dns-server 8.8.8.8
 domain-name lab-utp.pa
exit
!
ip dhcp excluded-address 192.168.2.81 192.168.2.85
ip dhcp pool VLAN20-SUC1
 network 192.168.2.80 255.255.255.240
 default-router 192.168.2.81
 dns-server 8.8.8.8
 domain-name lab-utp.pa
exit
!
! =========================================================
! OSPF AREA 0 - unico protocolo (no hay rutas estaticas)
! El respaldo a R3 tiene costo 100: el camino primario a la
! central queda por R1 y el enlace R2-R3 solo se usa si algo falla.
! =========================================================
router ospf 10
 router-id 192.168.2.17
 network 192.168.2.0 0.0.0.255 area 0
 passive-interface GigabitEthernet0/0
 passive-interface GigabitEthernet0/0.20
!
! =========================================================
! Alternativa RIPv2 (para comparar en la Parte 9)
! =========================================================
! no router ospf 10
! router rip
!  version 2
!  no auto-summary
!  network 192.168.2.0
!
end
copy running-config startup-config
